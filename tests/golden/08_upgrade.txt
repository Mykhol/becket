### an old-style settings.json: no $schema, no branchPrefix

# settings.json
{
  "repos": {
    "service": {
      "path": "./service",
      "defaultBase": "main"
    }
  }
}


### upgrade adds the schema reference and default keys

$ becket upgrade
[exit 0]
--- stdout ---
▸ Config may be outdated. Run becket upgrade to update.
▸ Updated settings.json

▸ Upgrade complete.

# settings.json
{
  "$schema": "./settings.schema.json",
  "repos": {
    "service": {
      "path": "./service",
      "defaultBase": "main"
    }
  },
  "branchPrefix": "feature/"
}


### upgrade is idempotent and also visits workspace manifests

$ becket upgrade
[exit 0]
--- stdout ---
▸ settings.json is already current.
▸ Workspace ws is already current.

▸ Upgrade complete.

# manifest: ws
{
  "$schema": "./workspace.schema.json",
  "id": "ws",
  "created": "<DATE>",
  "description": "",
  "repos": {
    "service": {
      "branch": "feature/ws",
      "base": "main"
    }
  }
}
